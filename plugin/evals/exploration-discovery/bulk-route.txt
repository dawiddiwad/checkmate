			if (pathname === '/api/bulk-reschedule' && request.method === 'POST') {
				const { changes } = await input(request)
				if (!Array.isArray(changes) || changes.length < 1 || changes.length > 10)
					return send(422, { error: 'Give 1–10 changes' })
				const updates = changes.map((change) => move(bookings.get(change.id), change))
				if (updates.some((item) => !item) || new Set(changes.map((change) => change.id)).size !== updates.length)
					return send(422, { error: 'The batch has an invalid or duplicate change' })
				for (const item of updates) {
					bookings.set(item.id, item)
					history.get(item.id).push({ action: 'rescheduled' })
				}
				return send(200, { items: updates })
			}
